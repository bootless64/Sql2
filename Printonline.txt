CREATE DATABASE PrintShopDB;
GO

USE PrintShopDB;
GO

CREATE TABLE Customers (
    CustomerID INT IDENTITY(1,1) PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Phone VARCHAR(20) NOT NULL,
    Email VARCHAR(100) NULL,
    CreatedAt DATETIME DEFAULT GETDATE()
);

CREATE TABLE Employees (
    EmployeeID INT IDENTITY(1,1) PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Phone VARCHAR(20),
    Role VARCHAR(50) NOT NULL,
    IsActive BIT DEFAULT 1
);

CREATE TABLE Services (
    ServiceID INT IDENTITY(1,1) PRIMARY KEY,
    ServiceName VARCHAR(50) NOT NULL,
    UnitType VARCHAR(20) NOT NULL
);

CREATE TABLE Prices (
    PriceID INT IDENTITY(1,1) PRIMARY KEY,
    ServiceID INT NOT NULL,
    PaperSize VARCHAR(20),
    PrintColor VARCHAR(10),
    PaperType VARCHAR(30),
    UnitPrice DECIMAL(10,2) NOT NULL,
    EffectiveFrom DATE NOT NULL,
    IsActive BIT DEFAULT 1,
    FOREIGN KEY (ServiceID) REFERENCES Services(ServiceID)
);

CREATE TABLE Orders (
    OrderID INT IDENTITY(1,1) PRIMARY KEY,
    CustomerID INT NOT NULL,
    CreatedByEmployeeID INT NOT NULL,
    OrderDateTime DATETIME DEFAULT GETDATE(),
    Status VARCHAR(20),
    Notes VARCHAR(300),
    TotalAmount DECIMAL(10,2) DEFAULT 0,
    DiscountAmount DECIMAL(10,2) DEFAULT 0,
    FinalAmount DECIMAL(10,2) DEFAULT 0,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID),
    FOREIGN KEY (CreatedByEmployeeID) REFERENCES Employees(EmployeeID)
);

CREATE TABLE OrderItems (
    OrderItemID INT IDENTITY(1,1) PRIMARY KEY,
    OrderID INT NOT NULL,
    ServiceID INT NOT NULL,
    Description VARCHAR(150),
    UnitPrice DECIMAL(10,2),
    LineTotal DECIMAL(10,2),
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID),
    FOREIGN KEY (ServiceID) REFERENCES Services(ServiceID)
);

CREATE TABLE Payments (
    PaymentID INT IDENTITY(1,1) PRIMARY KEY,
    OrderID INT NOT NULL,
    PaidAmount DECIMAL(10,2),
    PayMethod VARCHAR(20),
    PaidAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID)
);

CREATE TABLE FileAttachments (
    FileID INT IDENTITY(1,1) PRIMARY KEY,
    OrderItemID INT NOT NULL,
    FilePath VARCHAR(200),
    UploadedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (OrderItemID) REFERENCES OrderItems(OrderItemID)
);
